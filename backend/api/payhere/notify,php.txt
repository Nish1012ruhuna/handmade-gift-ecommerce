<?php

/*
 * =====================================================
 * GIFTORA - PAYHERE NOTIFICATION
 * =====================================================
 *
 * PayHere sends payment information to this file
 * after a payment attempt.
 *
 * This version records the notification so we can
 * test the Sandbox integration.
 *
 */


/*
 * -----------------------------------------------------
 * GET DATA FROM PAYHERE
 * -----------------------------------------------------
 */

$merchant_id =
    $_POST['merchant_id'] ?? '';


$order_id =
    $_POST['order_id'] ?? '';


$payhere_amount =
    $_POST['payhere_amount'] ?? '';


$payhere_currency =
    $_POST['payhere_currency'] ?? '';


$status_code =
    $_POST['status_code'] ?? '';


$md5sig =
    $_POST['md5sig'] ?? '';


/*
 * -----------------------------------------------------
 * CREATE DATA ARRAY
 * -----------------------------------------------------
 */

$data = [

    'merchant_id' =>
        $merchant_id,

    'order_id' =>
        $order_id,

    'amount' =>
        $payhere_amount,

    'currency' =>
        $payhere_currency,

    'status_code' =>
        $status_code,

    'md5sig' =>
        $md5sig,

    'received_at' =>
        date('Y-m-d H:i:s')

];


/*
 * -----------------------------------------------------
 * SAVE NOTIFICATION TO FILE
 * -----------------------------------------------------
 *
 * This is useful for testing your Sandbox integration.
 *
 */

$logFile =
    __DIR__ . '/payhere_notifications.txt';


file_put_contents(

    $logFile,

    json_encode(
        $data,
        JSON_PRETTY_PRINT
    ) .
    PHP_EOL .
    PHP_EOL,

    FILE_APPEND

);


/*
 * -----------------------------------------------------
 * RESPONSE
 * -----------------------------------------------------
 */

http_response_code(200);

echo "Notification received";

?>
